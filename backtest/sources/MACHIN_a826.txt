// Regression metrics 
get_historical_slope(float src, int start_index, int window_len) =>
    float sumX  = 0.0
    float sumY  = 0.0
    float sumXY = 0.0
    float sumX2 = 0.0

    for i = start_index to (start_index + window_len - 1)
        float x = i // Bar index or relative index
        float y = src[i]
        sumX  += x
        sumY  += y
        sumXY += x * y
        sumX2 += x * x
        
    float slope = (window_len * sumXY - sumX * sumY) / (window_len * sumX2 - math.pow(sumX, 2))
    slope

// KNN ML 
f_knn_k4_dist(float current_val, float[] history_array) =>
    float avg_dist = 1000000.0
    int k = knn_clusters
    if array.size(history_array) >= k
        float[] distances = array.new_float(0)
        for i = 0 to array.size(history_array) - 1
            array.push(distances, math.abs(current_val - array.get(history_array, i)))
        array.sort(distances, order.ascending)
        

        float sum = 0.0
        for i = 0 to k - 1
            sum += array.get(distances, i)
        avg_dist := sum / k
    else if array.size(history_array) > 0
        avg_dist := math.abs(current_val - array.get(history_array, 0))
    
    avg_dist
