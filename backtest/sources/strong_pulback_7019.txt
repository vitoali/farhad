f_size(string s) =>
    switch s
        "Tiny"   => size.tiny
        "Small"  => size.small
        "Normal" => size.normal
        "Large"  => size.large
        => size.small

f_pos(string p) =>
    switch p
        "Top Left"     => position.top_left
        "Top Right"    => position.top_right
        "Bottom Left"  => position.bottom_left
        "Bottom Right" => position.bottom_right
        "Middle Right" => position.middle_right
        => position.top_right

f_stars(float sc) => sc >= 8.5 ? "★★★★★" : sc >= 7 ? "★★★★" : sc >= 5.5 ? "★★★" : sc >= 4 ? "★★" : sc >= 2.5 ? "★" : "·"
f_tier(float sc)  => sc >= 8.5 ? "ELITE" : sc >= 7 ? "VERY STRONG" : sc >= 5.5 ? "STRONG" : sc >= 4 ? "MODERATE" : sc >= 2.5 ? "WEAK" : "FORMING"
f_signedR(float v) => (v >= 0 ? "+" : "") + str.tostring(v, "#.##") + "R"

f_inList(string csv, int hr) =>
    bool found = false
    arr = str.split(csv, ",")
    if array.size(arr) > 0
        for s in arr
            if str.tonumber(str.trim(s)) == hr
                found := true
    found

f_score(int side) =>
    float a   = atr <= 0 ? syminfo.mintick : atr
    float htfF= not useHtf or na(htfEma) ? 0.6 : ((side == 1 ? close > htfEma : close < htfEma) ? 1.0 : 0.0)
    float sesF= inWeakHour ? 0.0 : 1.0
    float sepF= math.min(math.abs(fastEma - slowEma) / (a * sepNorm), 1.0)
    float depF= math.min(entryDepth / 0.6, 1.0)
    float momF= side == 1 ? math.min(math.max((rsi - 50.0) / 30.0, 0.0), 1.0) : math.min(math.max((50.0 - rsi) / 30.0, 0.0), 1.0)
    float volF= volSma > 0 ? math.min(volume / volSma, 1.0) : 0.0
    float wsum= wHtf + wSess + wTrend + wDepth + wMom + wVol
    float raw = wsum > 0 ? (htfF * wHtf + sesF * wSess + sepF * wTrend + depF * wDepth + momF * wMom + volF * wVol) / wsum : 0.0
    math.min(math.max(raw * 10.0, 0.0), 10.0)

f_tooltip(int side, float sc) =>
    string dir = side == 1 ? "LONG" : "SHORT"
    string tt  = "Setup quality: " + str.tostring(math.round(sc * 10) / 10) + "/10  " + f_stars(sc) + "  " + f_tier(sc)
    tt += "\nDirection: " + dir
    tt += "\nHTF aligned: " + (not useHtf ? "filter off" : na(htfEma) ? "n/a" : ((side == 1 ? close > htfEma : close < htfEma) ? "yes" : "NO"))
    tt += "\nTrend sep/ATR: " + str.tostring(atr > 0 ? math.abs(fastEma - slowEma) / atr : 0.0, "#.##")
    tt += "\nEntry depth (ATR): " + str.tostring(entryDepth, "#.##")
    tt += "\nNote: score is a confluence read, not a win-rate guarantee."
    tt
