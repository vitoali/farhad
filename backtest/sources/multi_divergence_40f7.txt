f_div_type(array<int> osc_hi_off, array<float> osc_hi_val, array<int> osc_lo_off, array<float> osc_lo_val) =>
    int result = 0
    if array.size(osc_hi_off) >= 2 and array.size(pr_hi_off) >= 2
        float o1 = array.get(osc_hi_val, 0)
        float o2 = array.get(osc_hi_val, 1)
        float p1 = array.get(pr_hi_val,  0)
        float p2 = array.get(pr_hi_val,  1)
        if p1 > p2 and o1 < o2
            result := -1
    if array.size(osc_lo_off) >= 2 and array.size(pr_lo_off) >= 2
        float o1 = array.get(osc_lo_val, 0)
        float o2 = array.get(osc_lo_val, 1)
        float p1 = array.get(pr_lo_val,  0)
        float p2 = array.get(pr_lo_val,  1)
        if p1 < p2 and o1 > o2
            result := 1
    result

    if array.size(osc_lo) >= 2 and array.size(s_pr_lo) >= 2
        if array.get(s_pr_lo, 0) < array.get(s_pr_lo, 1) and array.get(osc_lo, 0) > array.get(osc_lo, 1)
            result := 1
    result

f_resolve_open_trades() =>
    int n = array.size(perf_status)
    if n > 0
        for i = 0 to n - 1
            if array.get(perf_status, i) == 0
                int   dir  = array.get(perf_direction, i)
                float tp_p = array.get(perf_tp, i)
                float sl_p = array.get(perf_sl, i)
                int   ind  = array.get(perf_ind_id, i)
                bool  hit_tp = dir == 1 ? high >= tp_p : low  <= tp_p
                bool  hit_sl = dir == 1 ? low  <= sl_p : high >= sl_p
                if hit_tp or hit_sl
                    array.set(perf_status, i, hit_tp ? 1 : -1)
                    int   cur_wins  = array.get(ind_wins,   ind)
                    int   cur_loss  = array.get(ind_losses, ind)
                    float cur_pnl   = array.get(ind_cum_pnl_atr, ind)
                    if hit_tp
                        array.set(ind_wins, ind, cur_wins + 1)
                        float aw = array.get(ind_avg_win_atr, ind)
                        array.set(ind_avg_win_atr, ind, (aw * cur_wins + tp_atr_mult) / (cur_wins + 1))
                        array.set(ind_cum_pnl_atr, ind, cur_pnl + tp_atr_mult)
                    else
                        array.set(ind_losses, ind, cur_loss + 1)
                        float al = array.get(ind_avg_loss_atr, ind)
                        array.set(ind_avg_loss_atr, ind, (al * cur_loss + sl_atr_mult) / (cur_loss + 1))
                        array.set(ind_cum_pnl_atr, ind, cur_pnl - sl_atr_mult)
