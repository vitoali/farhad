// Picks the marker symbol and which way it points.
markerChar(string style, bool fromBelow) =>
    string result = "●"
    if style == "Triangle"
        result := fromBelow ? "▼" : "▲"
    else if style == "Circle"
        result := "●"
    else if style == "Cross"
        result := "✖"
    else if style == "Diamond"
        result := "◆"
    result

// Converts a size dropdown choice into a Pine size constant.
sizeFromStr(string s) =>
    switch s
        "Tiny" => size.tiny
        "Small" => size.small
        "Normal" => size.normal
        "Large" => size.large
        "Huge" => size.huge
        => size.small

// Builds the top and bottom of a zone from the swing, padded by the average wick.
getZoneBounds(int barsBack, bool isSupply) =>
    float pivotH = high[barsBack]
    float pivotL = low[barsBack]
    float sumWick = 0.0
    for k = 0 to WICK_AVG_BARS - 1
        float h = high[barsBack + k]
        float l = low[barsBack + k]
        float o = open[barsBack + k]
        float c = close[barsBack + k]
        sumWick += isSupply ? (h - math.max(o, c)) : (math.min(o, c) - l)
    float avgWick = sumWick / WICK_AVG_BARS
    float top = isSupply ? pivotH : pivotL + avgWick
    float bot = isSupply ? pivotH - avgWick : pivotL
    [top, bot]

// Keeps the zone list from growing forever.
trimArray(array<Zone> zones) =>
    if array.size(zones) > MAX_ZONES
        array.shift(zones)

// Checks if a new zone would sit on top of an existing one.
hasOverlap(float newTop, float newBottom, array<Zone> zones) =>
    bool overlap = false
    if array.size(zones) > 0
        for i = 0 to array.size(zones) - 1
            Zone z = array.get(zones, i)
            bool ageOk = bar_index - z.createdBar <= lookbackBars
            if z.active and ageOk and newBottom <= z.top and newTop >= z.bottom
                overlap := true
                break
    overlap

// Checks each zone for new retests, breakouts and flips on this bar.
updateZones(array<Zone> zones, EventFlags evts) =>
    if array.size(zones) > 0
        for i = 0 to array.size(zones) - 1
            Zone z = array.get(zones, i)
            if z.active and bar_index - z.createdBar > lookbackBars
                z.active := false
            if z.active
                bool brokeThrough = (z.isSupply and close > z.top) or (not z.isSupply and close < z.bottom)
                bool brokeBack = (z.isSupply and close < z.bottom) or (not z.isSupply and close > z.top)
                bool flipped = false
                bool pendingHandled = false
                if z.pendingFlip and brokeThrough
                    z.breakouts += 1
                    array.push(z.breakoutTimes, time)
                    array.push(z.breakoutFromBelow, z.isSupply)
                    array.push(z.breakoutBars, bar_index)
                    if z.isSupply
                        evts.bullBreakout := true
                    else
                        evts.bearBreakout := true
                    z.isSupply := not z.isSupply
                    flipped := true
                    if not na(z.segments) and array.size(z.segments) > 0
                        Segment lastSeg = array.get(z.segments, array.size(z.segments) - 1)
                        lastSeg.endTime := time
                    array.push(z.segments, Segment.new(startTime = time, isSupply = z.isSupply))
                    z.pendingFlip := false
                    pendingHandled := true
                if z.pendingFlip and brokeBack
                    z.retests += 1
                    z.lastRetestBar := bar_index
                    array.push(z.retestTimes, time)
                    array.push(z.retestFromBelow, z.isSupply)
                    array.push(z.retestPrices, z.isSupply ? high : low)
                    array.push(z.retestBars, bar_index)
                    if z.isSupply
                        evts.bearRetest := true
                    else
                        evts.bullRetest := true
                    z.pendingFlip := false
                    pendingHandled := true
                if not pendingHandled and not z.pendingFlip and brokeThrough
                    z.pendingFlip := true
                if z.active and not flipped and not z.pendingFlip and not pendingHandled and bar_index > z.createdBar and bar_index - z.lastRetestBar >= cooldownBars
                    bool reacted = false
                    if z.isSupply
                        if high >= z.bottom and close < z.bottom
                            reacted := true
                    else
                        if low <= z.top and close > z.top
                            reacted := true
                    if reacted
                        z.retests += 1
                        z.lastRetestBar := bar_index
                        array.push(z.retestTimes, time)
                        array.push(z.retestFromBelow, z.isSupply)
                        array.push(z.retestPrices, z.isSupply ? high : low)
                        array.push(z.retestBars, bar_index)
                        if z.isSupply
                            evts.bearRetest := true
                        else
                            evts.bullRetest := true

// Is this zone already in the list?
zoneInArr(array<Zone> arr, Zone target) =>
    bool found = false
    if array.size(arr) > 0
        for i = 0 to array.size(arr) - 1
            if array.get(arr, i).id == target.id
                found := true
                break
    found

// Counts how many retests held: price came back, then the zone wasn't broken within the
// Hold Window. Only counts retests old enough to have a full window behind them.
// Returns [held, checked].
computeHoldStats(Zone z) =>
    int hits = 0
    int resolved = 0
    if not na(z.retestBars) and array.size(z.retestBars) > 0
        for ri = 0 to array.size(z.retestBars) - 1
            int rb = array.get(z.retestBars, ri)
            if last_bar_index - rb >= holdBars
                resolved += 1
                bool broke = false
                if not na(z.breakoutBars) and array.size(z.breakoutBars) > 0
                    for bi = 0 to array.size(z.breakoutBars) - 1
                        int bb = array.get(z.breakoutBars, bi)
                        if bb > rb and bb <= rb + holdBars
                            broke := true
                            break
                if not broke
                    hits += 1
    [hits, resolved]

// Gives each zone a score for ranking.
//  - Nearest: just how close it is to price.
//  - Strongest: combines how reliably the zone held after retests, how many times it's been
//    tested, and how close it is to price. A zone already on screen gets a small edge so the
//    shown list stays steady instead of swapping every bar.
scoreZone(Zone z) =>
    float refClose = nz(lastConfirmedClose, close)
    float mid = (z.top + z.bottom) / 2.0
    float dist = math.abs(mid - refClose)
    float result = na
    if rankBy == "Nearest"
        result := -dist
    else
        float relDist = refClose > 0 ? dist / refClose : dist
        [hits, resolved] = computeHoldStats(z)
        float quality = (hits + HOLD_PRIOR_HITS) / (resolved + HOLD_PRIOR_TOTAL)
        float evidence = 1.0 + math.log(1 + z.retests)
        float proximity = 1.0 / (1.0 + relDist * DIST_SENS)
        float core = quality * evidence * proximity
        bool incumbent = zoneInArr(prevVisibleSupply, z) or zoneInArr(prevVisibleDemand, z)
        result := incumbent ? core * (1.0 + INCUMBENT_BONUS) : core
    result

// Clears all drawings so we can redraw fresh.
clearManaged() =>
    if array.size(managedBoxes) > 0
        for i = 0 to array.size(managedBoxes) - 1
            box.delete(array.get(managedBoxes, i))
        array.clear(managedBoxes)
    if array.size(managedLabels) > 0
        for i = 0 to array.size(managedLabels) - 1
            label.delete(array.get(managedLabels, i))
        array.clear(managedLabels)
    if array.size(managedLines) > 0
        for i = 0 to array.size(managedLines) - 1
            line.delete(array.get(managedLines, i))
        array.clear(managedLines)

// Gathers the zones that are still valid and on the right side of price.
collectCandidates(array<Zone> source, array<Zone> supplyOut, array<Zone> demandOut) =>
    float refClose = nz(lastConfirmedClose, close)
    if array.size(source) > 0
        for i = 0 to array.size(source) - 1
            Zone z = array.get(source, i)
            bool ageOk = bar_index - z.createdBar <= lookbackBars
            bool sideOk = z.pendingFlip or (z.isSupply ? refClose <= z.top : refClose >= z.bottom)
            if z.active and ageOk and sideOk
                if z.isSupply
                    array.push(supplyOut, z)
                else
                    array.push(demandOut, z)

// Finds a fading zone in the fade list.
fadeIdxOf(array<FadeEntry> arr, Zone target) =>
    int result = -1
    if array.size(arr) > 0
        for i = 0 to array.size(arr) - 1
            if array.get(arr, i).zone.id == target.id
                result := i
                break
    result

// Is this zone too close to one we've already chosen?
isTooClose(float zMid, array<float> pickedMids, float refClose) =>
    bool result = false
    int pickedSize = array.size(pickedMids)
    if minZoneSpacing > 0 and pickedSize > 0
        for k = 0 to pickedSize - 1
            float pzMid = array.get(pickedMids, k)
            float dist = math.abs(zMid - pzMid)
            float relDist = refClose > 0 ? dist / refClose : dist
            if relDist < minZoneSpacing
                result := true
                break
    result

// Picks the top zones to show on each side, ranked by score.
selectVisible(array<Zone> supplyCandidates, array<Zone> demandCandidates, array<Zone> supplyOut, array<Zone> demandOut) =>
    float refClose = nz(lastConfirmedClose, close)
    int supplyN = array.size(supplyCandidates)
    int demandN = array.size(demandCandidates)
    int supplyCap = math.min(showNearest, supplyN)
    int demandCap = math.min(showNearest, demandN)
    int maxPicks = supplyCap + demandCap
    if maxPicks > 0
        array<float> supplyScores = array.new<float>()
        array<float> supplyMids = array.new<float>()
        if supplyN > 0
            for i = 0 to supplyN - 1
                Zone z = array.get(supplyCandidates, i)
                array.push(supplyScores, scoreZone(z))
                array.push(supplyMids, (z.top + z.bottom) / 2.0)
        array<float> demandScores = array.new<float>()
        array<float> demandMids = array.new<float>()
        if demandN > 0
            for i = 0 to demandN - 1
                Zone z = array.get(demandCandidates, i)
                array.push(demandScores, scoreZone(z))
                array.push(demandMids, (z.top + z.bottom) / 2.0)
        array<bool> supplyUsed = array.new<bool>(supplyN, false)
        array<bool> demandUsed = array.new<bool>(demandN, false)
        array<float> pickedMids = array.new<float>()
        for n = 0 to maxPicks - 1
            int bestSupplyIdx = -1
            int bestDemandIdx = -1
            float bestScore = -1e20
            bool bestIsSupply = false
            if array.size(supplyOut) < supplyCap and supplyN > 0
                for i = 0 to supplyN - 1
                    if not array.get(supplyUsed, i)
                        if not isTooClose(array.get(supplyMids, i), pickedMids, refClose)
                            float s = array.get(supplyScores, i)
                            if s > bestScore
                                bestScore := s
                                bestSupplyIdx := i
                                bestIsSupply := true
            if array.size(demandOut) < demandCap and demandN > 0
                for i = 0 to demandN - 1
                    if not array.get(demandUsed, i)
                        if not isTooClose(array.get(demandMids, i), pickedMids, refClose)
                            float s = array.get(demandScores, i)
                            if s > bestScore
                                bestScore := s
                                bestDemandIdx := i
                                bestIsSupply := false
            if bestIsSupply and bestSupplyIdx >= 0
                array.set(supplyUsed, bestSupplyIdx, true)
                Zone bestZone = array.get(supplyCandidates, bestSupplyIdx)
                array.push(supplyOut, bestZone)
                array.push(pickedMids, array.get(supplyMids, bestSupplyIdx))
            else if not bestIsSupply and bestDemandIdx >= 0
                array.set(demandUsed, bestDemandIdx, true)
                Zone bestZone = array.get(demandCandidates, bestDemandIdx)
                array.push(demandOut, bestZone)
                array.push(pickedMids, array.get(demandMids, bestDemandIdx))
            else
                break

// Draws one zone: the colored box, its markers and its stats label.
renderOneZone(Zone z, int steps, float fadeLevel) =>
    int futureTime = time + timeframe.in_seconds() * 1000 * 10
    int edgeTransp = math.min(100, int(math.round(100 * fadeLevel)))
    color zoneColor = z.isSupply ? supplyBase : demandBase
    color statusCol = color.new(zoneColor, edgeTransp)
    float lightTransp = ZONE_TRANSPARENCY
    float deepTransp = math.max(0.0, ZONE_TRANSPARENCY - GRADIENT_DEPTH)
    if not na(z.segments) and array.size(z.segments) > 0
        for j = 0 to array.size(z.segments) - 1
            Segment seg = array.get(z.segments, j)
            bool segActive = seg.endTime == 0
            int segRight = segActive ? futureTime : seg.endTime
            color segBaseColor = seg.isSupply ? supplyBase : demandBase
            color segEdgeCol = color.new(segBaseColor, edgeTransp)
            float transpBottom = seg.isSupply ? lightTransp : deepTransp
            float transpTop = seg.isSupply ? deepTransp : lightTransp
            float sliceH = (z.top - z.bottom) / steps
            for s = 0 to steps - 1
                float sliceBot = z.bottom + sliceH * s
                float sliceTop = z.bottom + sliceH * (s + 1)
                float f = (s + 0.5) / steps
                float sliceTransp = transpBottom + (transpTop - transpBottom) * f
                float fadedTransp = sliceTransp + (100.0 - sliceTransp) * fadeLevel
                array.push(managedBoxes, box.new(left = seg.startTime, top = sliceTop, right = segRight, bottom = sliceBot, bgcolor = color.new(segBaseColor, int(math.round(fadedTransp))), border_color = na, xloc = xloc.bar_time))
            float baseY = seg.isSupply ? z.top : z.bottom
            array.push(managedLines, line.new(x1 = seg.startTime, y1 = baseY, x2 = segRight, y2 = baseY, xloc = xloc.bar_time, color = segEdgeCol, width = 1))
    if showRetests and not na(z.retestTimes) and array.size(z.retestTimes) > 0
        for j = 0 to array.size(z.retestTimes) - 1
            int rt = array.get(z.retestTimes, j)
            bool fromBelow = array.get(z.retestFromBelow, j)
            float rp = array.get(z.retestPrices, j)
            float markerY = fromBelow ? math.max(rp, z.top) : math.min(rp, z.bottom)
            color baseMarkerCol = fromBelow ? supplyShine : demandShine
            string glyph = markerChar(retestMarker, fromBelow)
            if MARKER_GLOW
                color glowCol = color.new(baseMarkerCol, math.min(100, edgeTransp + 45))
                array.push(managedLabels, label.new(x = rt, y = markerY, text = glyph, xloc = xloc.bar_time, yloc = yloc.price, style = label.style_label_center, color = NONE_COLOR, textcolor = glowCol, size = size.large))
            color coreCol = color.new(baseMarkerCol, edgeTransp)
            array.push(managedLabels, label.new(x = rt, y = markerY, text = glyph, xloc = xloc.bar_time, yloc = yloc.price, style = label.style_label_center, color = NONE_COLOR, textcolor = coreCol, size = size.small))
    if showBreakouts and not na(z.breakoutTimes) and array.size(z.breakoutTimes) > 0
        for j = 0 to array.size(z.breakoutTimes) - 1
            int bt = array.get(z.breakoutTimes, j)
            bool fromBelow = array.get(z.breakoutFromBelow, j)
            color markerCol = color.new(fromBelow ? supplyBase : demandBase, edgeTransp)
            array.push(managedLabels, label.new(x = bt, y = 0, text = markerChar(breakoutMarker, fromBelow), xloc = xloc.bar_time, yloc = fromBelow ? yloc.abovebar : yloc.belowbar, style = label.style_label_center, color = NONE_COLOR, textcolor = markerCol, size = size.small))
    // Build the stats label: retests, breakouts, and how many of those retests held.
    if showStats
        [holdHits, holdResolved] = computeHoldStats(z)
        string holdTxt = z.retests > 0 ? " · Held: " + str.tostring(holdHits) + " (" + str.tostring(100.0 * holdHits / z.retests, "#") + "%)" : ""
        string statusTxt = "Retests: " + str.tostring(z.retests) + " · Breakouts: " + str.tostring(z.breakouts) + holdTxt
        bool topBottom = statsPosition == "Top & Bottom"
        float statsY = topBottom ? (z.isSupply ? z.top : z.bottom) : (z.top + z.bottom) / 2.0
        string statsStyle = topBottom ? (z.isSupply ? label.style_label_lower_right : label.style_label_upper_right) : (statsPosition == "Right Center" ? label.style_label_left : label.style_label_right)
        array.push(managedLabels, label.new(x = futureTime, y = statsY, text = statusTxt, xloc = xloc.bar_time, yloc = yloc.price, style = statsStyle, color = NONE_COLOR, textcolor = statusCol, size = sizeFromStr(statsSize)))

// Adds up the boxes needed so drawing stays under the limit.
countSegments(array<Zone> zones) =>
    int total = 0
    if array.size(zones) > 0
        for i = 0 to array.size(zones) - 1
            Zone z = array.get(zones, i)
            total += na(z.segments) ? 1 : math.max(1, array.size(z.segments))
    total

// Refreshes which zones show; ones that drop off start fading out.
trackVisibility() =>
    array<Zone> supplyCandidates = array.new<Zone>()
    array<Zone> demandCandidates = array.new<Zone>()
    collectCandidates(supplyZones, supplyCandidates, demandCandidates)
    collectCandidates(demandZones, supplyCandidates, demandCandidates)
    array<Zone> newVisibleSupply = array.new<Zone>()
    array<Zone> newVisibleDemand = array.new<Zone>()
    selectVisible(supplyCandidates, demandCandidates, newVisibleSupply, newVisibleDemand)
    if array.size(prevVisibleSupply) > 0
        for i = 0 to array.size(prevVisibleSupply) - 1
            Zone z = array.get(prevVisibleSupply, i)
            if not zoneInArr(newVisibleSupply, z) and not zoneInArr(newVisibleDemand, z) and fadeIdxOf(fadingZones, z) < 0
                array.push(fadingZones, FadeEntry.new(zone = z, startBar = bar_index))
    if array.size(prevVisibleDemand) > 0
        for i = 0 to array.size(prevVisibleDemand) - 1
            Zone z = array.get(prevVisibleDemand, i)
            if not zoneInArr(newVisibleSupply, z) and not zoneInArr(newVisibleDemand, z) and fadeIdxOf(fadingZones, z) < 0
                array.push(fadingZones, FadeEntry.new(zone = z, startBar = bar_index))
    if array.size(fadingZones) > 0
        for i = array.size(fadingZones) - 1 to 0
            FadeEntry fe = array.get(fadingZones, i)
            bool cameBack = zoneInArr(newVisibleSupply, fe.zone) or zoneInArr(newVisibleDemand, fe.zone)
            bool expired = bar_index - fe.startBar >= FADE_BARS
            if cameBack or expired
                array.remove(fadingZones, i)
    array.clear(prevVisibleSupply)
    array.clear(prevVisibleDemand)
    if array.size(newVisibleSupply) > 0
        for i = 0 to array.size(newVisibleSupply) - 1
            array.push(prevVisibleSupply, array.get(newVisibleSupply, i))
    if array.size(newVisibleDemand) > 0
        for i = 0 to array.size(newVisibleDemand) - 1
            array.push(prevVisibleDemand, array.get(newVisibleDemand, i))

// Draws the current zones solid and the fading ones dimmed.
drawTop() =>
    int segs = countSegments(prevVisibleSupply) + countSegments(prevVisibleDemand)
    if array.size(fadingZones) > 0
        for i = 0 to array.size(fadingZones) - 1
            FadeEntry fe = array.get(fadingZones, i)
            segs += na(fe.zone.segments) ? 1 : math.max(1, array.size(fe.zone.segments))
    int steps = segs > 0 ? math.max(1, math.min(GRADIENT_STEPS, int(BOX_BUDGET / segs))) : GRADIENT_STEPS
    if array.size(prevVisibleSupply) > 0
        for i = 0 to array.size(prevVisibleSupply) - 1
            renderOneZone(array.get(prevVisibleSupply, i), steps, 0.0)
    if array.size(prevVisibleDemand) > 0
        for i = 0 to array.size(prevVisibleDemand) - 1
            renderOneZone(array.get(prevVisibleDemand, i), steps, 0.0)
    if array.size(fadingZones) > 0
        for i = 0 to array.size(fadingZones) - 1
            FadeEntry fe = array.get(fadingZones, i)
            float fadeLevel = (float(bar_index - fe.startBar) + 1) / float(FADE_BARS)
            renderOneZone(fe.zone, steps, fadeLevel)
//#endregion FUNCTION DECLARATIONS

//#region CALCULATIONS
// Look for a new swing high and swing low.
float ph = ta.pivothigh(high, pivotLen, pivotLen)
float pl = ta.pivotlow(low, pivotLen, pivotLen)

// New swing high makes a supply zone (unless it clashes with an existing one).
if not na(ph)
    [t, b] = getZoneBounds(pivotLen, true)
    bool overlap = hasOverlap(t, b, supplyZones) or hasOverlap(t, b, demandZones)
    if not na(t) and not na(b) and not overlap
        Zone newZone = Zone.new(id = zoneIdCounter, top = t, bottom = b, createdBar = bar_index - pivotLen, createdTime = time[pivotLen], isSupply = true, retestTimes = array.new<int>(), retestFromBelow = array.new<bool>(), retestPrices = array.new<float>(), retestBars = array.new<int>(), breakoutTimes = array.new<int>(), breakoutFromBelow = array.new<bool>(), breakoutBars = array.new<int>(), segments = array.new<Segment>())
        zoneIdCounter += 1
        array.push(newZone.segments, Segment.new(startTime = time[pivotLen], isSupply = true))
        array.push(supplyZones, newZone)
        trimArray(supplyZones)

// New swing low makes a demand zone (unless it clashes with an existing one).
if not na(pl)
    [t, b] = getZoneBounds(pivotLen, false)
    bool overlap = hasOverlap(t, b, supplyZones) or hasOverlap(t, b, demandZones)
    if not na(t) and not na(b) and not overlap
        Zone newZone = Zone.new(id = zoneIdCounter, top = t, bottom = b, createdBar = bar_index - pivotLen, createdTime = time[pivotLen], isSupply = false, retestTimes = array.new<int>(), retestFromBelow = array.new<bool>(), retestPrices = array.new<float>(), retestBars = array.new<int>(), breakoutTimes = array.new<int>(), breakoutFromBelow = array.new<bool>(), breakoutBars = array.new<int>(), segments = array.new<Segment>())
        zoneIdCounter += 1
        array.push(newZone.segments, Segment.new(startTime = time[pivotLen], isSupply = false))
        array.push(demandZones, newZone)
        trimArray(demandZones)

EventFlags evts = EventFlags.new()

// Once a bar closes, update zones and refresh what's shown.
if barstate.isconfirmed
    lastConfirmedClose := close
    updateZones(supplyZones, evts)
    updateZones(demandZones, evts)
    if last_bar_index - bar_index <= FADE_BARS + 1
        trackVisibility()
//#endregion CALCULATIONS

//#region VISUALS
// Redraw everything on the latest bar.
if barstate.islast
    clearManaged()
    drawTop()
//#endregion VISUALS
