sigNeedHtf  = input.bool(true,  "Require HTF agreement",   group=gSig,

gDash = "Dashboard / info table"

// =====================================================================
//                              STATE
// =====================================================================
var float lastPH    = na
var float lastPL    = na
var int   lastPHbar = na
var int   lastPLbar = na
var float prevPH    = na
var float prevPL    = na
var int   prevPHbar = na
var int   prevPLbar = na
var bool  phBroken  = true
var bool  plBroken  = true
var int   trend     = 0
var string lastSig  = "-"      // last confluence signal for the dashboard
 
var box[] bullOBs      = array.new<box>()
var box[] bearOBs      = array.new<box>()
var box[] bullBreakers = array.new<box>()
var box[] bearBreakers = array.new<box>()
var box[] bullFVGs     = array.new<box>()
var box[] bearFVGs     = array.new<box>()
 
// Premium/Discount objects - reused
var box  pdBoxPrem = na
var box  pdBoxDisc = na
var line pdLineEq  = na
var line pdLineTop = na
var line pdLineBot = na
 
atr14  = ta.atr(14)
volSma = ta.sma(volume, 20)
strongVolNow = not na(volSma) and volume > volSma * volMult
 
// =====================================================================
//                              HTF BIAS
// =====================================================================
// Non-biased, non-repainting request: take the confirmed value (expr[1])
// from the higher timeframe. The request.security call is in global scope.
// Pine v6 note: bool series can no longer be na, so the HTF state is
// requested as 1/0 and converted back to bool. Same non-repainting logic.
htfUpRaw = request.security(syminfo.tickerid, htfRes,
              (ta.ema(close, htfFast) > ta.ema(close, htfSlow))[1] ? 1 : 0,
              lookahead = barmerge.lookahead_on)
bool htfUp = nz(htfUpRaw) == 1
htfBull = useHtf and htfUp
htfBear = useHtf and not htfUp
htfBiasTxt = not useHtf ? "OFF" : htfBull ? "BULL" : htfBear ? "BEAR" : "-"
htfBiasCol = not useHtf ? color.gray : htfBull ? bullCol : htfBear ? bearCol : color.gray
 
// =====================================================================
//                              SWINGS
// =====================================================================
ph = ta.pivothigh(high, swingLen, swingLen)
pl = ta.pivotlow(low,  swingLen, swingLen)
 
// Flags for plotshape - pivot event on the current bar
bool didEQH = false
bool didEQL = false
bool didH   = false
bool didL   = false
 
if not na(ph)
    if showEQ and not na(lastPH) and math.abs(ph - lastPH) <= atr14 * eqTolAtr
        didEQH := true
        line.new(lastPHbar, lastPH, bar_index - swingLen, ph, color=eqCol, style=line.style_dotted, width=2)
        if aEQ
            alert("EQH: equal highs - liquidity zone above", alert.freq_once_per_bar)
    else
        didH := true
    prevPH    := lastPH
    prevPHbar := lastPHbar
    lastPH    := ph
    lastPHbar := bar_index - swingLen
    phBroken  := false
 
if not na(pl)
    if showEQ and not na(lastPL) and math.abs(pl - lastPL) <= atr14 * eqTolAtr
        didEQL := true
        line.new(lastPLbar, lastPL, bar_index - swingLen, pl, color=eqCol, style=line.style_dotted, width=2)
        if aEQ
            alert("EQL: equal lows - liquidity zone below", alert.freq_once_per_bar)
    else
        didL := true
    prevPL    := lastPL
    prevPLbar := lastPLbar
    lastPL    := pl
    lastPLbar := bar_index - swingLen
    plBroken  := false
 
// Strictly anchored swing and EQH/EQL markers - they do not overlap
plotshape(showSwings and not compactMode and didH,
  title    = "Swing High",
  style    = shape.circle,
  location = location.abovebar,
  color    = color.new(bearCol, 50),
  size     = size.tiny,
  offset   = -swingLen)
plotshape(showSwings and not compactMode and didL,
  title    = "Swing Low",
  style    = shape.circle,
  location = location.belowbar,
  color    = color.new(bullCol, 50),
  size     = size.tiny,
  offset   = -swingLen)
plotshape(showEQ and not compactMode and didEQH,
  title    = "EQH",
  style    = shape.diamond,
  location = location.abovebar,
  color    = color.new(eqCol, 20),
  size     = size.tiny,
  offset   = -swingLen)
plotshape(showEQ and not compactMode and didEQL,
  title    = "EQL",
  style    = shape.diamond,
  location = location.belowbar,
  color    = color.new(eqCol, 20),
  size     = size.tiny,
  offset   = -swingLen)
 
// =====================================================================
//                          LIQUIDITY SWEEPS
// =====================================================================
// Sweep above (bearish sweep - buy-side liquidity taken):
//   high broke lastPH but close returned below lastPH
bearSweepRaw = showLiq and not na(lastPH) and not phBroken and high > lastPH and close < lastPH
bullSweepRaw = showLiq and not na(lastPL) and not plBroken and low  < lastPL and close > lastPL
// Optional volume filter
bearSweep = bearSweepRaw and (not liqVolConf or strongVolNow)
bullSweep = bullSweepRaw and (not liqVolConf or strongVolNow)
 
if bearSweep and aSweep
    alert("Liquidity sweep above (high grab)", alert.freq_once_per_bar)
if bullSweep and aSweep
    alert("Liquidity sweep below (low grab)", alert.freq_once_per_bar)
 
// Sweep signals as plotshape - fixed position at the candle
plotshape(bearSweep and not compactMode,
  title    = "Sweep High",
  style    = shape.triangledown,
  location = location.abovebar,
  color    = bearLiqCol,
  size     = size.small)
plotshape(bullSweep and not compactMode,
  title    = "Sweep Low",
  style    = shape.triangleup,
  location = location.belowbar,
  color    = bullLiqCol,
  size     = size.small)
plotchar(bearSweep and not compactMode and liqLabel == "Text",
  title    = "Sweep Text Top",
  char     = "S",
  location = location.abovebar,
  color    = bearLiqCol,
  size     = size.tiny)
plotchar(bullSweep and not compactMode and liqLabel == "Text",
  title    = "Sweep Text Bot",
  char     = "S",
  location = location.belowbar,
  color    = bullLiqCol,
  size     = size.tiny)
 
// =====================================================================
//                          BOS / CHoCH
// =====================================================================
bullEvent = not na(lastPH) and not phBroken and close > lastPH
bearEvent = not na(lastPL) and not plBroken and close < lastPL
 
isBullCHoCH = bullEvent and trend == -1
isBullBOS   = bullEvent and trend != -1
isBearCHoCH = bearEvent and trend ==  1
isBearBOS   = bearEvent and trend !=  1
 
if bullEvent
    phBroken := true
    trend    := 1
if bearEvent
    plBroken := true
    trend    := -1
 
drawStructure(level, fromBar, isBull, isCH) =>
    col = isBull ? bullCol : bearCol
    txt = isCH ? " CHoCH" : " BOS"
    line.new(fromBar, level, bar_index, level, color=col, style=line.style_dashed, width=1)
    // style_none = text only, no bubble. Hard-anchored to (bar_index, price).
    label.new(bar_index, level, txt,
      style     = label.style_none,
      textcolor = col,
      size      = size.small)
 
if (isBullBOS and showBOS) or (isBullCHoCH and showCHoCH)
    drawStructure(lastPH, lastPHbar, true,  isBullCHoCH)
if (isBearBOS and showBOS) or (isBearCHoCH and showCHoCH)
    drawStructure(lastPL, lastPLbar, false, isBearCHoCH)
 
if aBOS and (isBullBOS or isBearBOS)
    alert("BOS: " + (isBullBOS ? "bullish" : "bearish") + " break of structure", alert.freq_once_per_bar)
if aCHOCH and (isBullCHoCH or isBearCHoCH)
    alert("CHoCH: change of character to the " + (isBullCHoCH ? "bullish" : "bearish") + " side", alert.freq_once_per_bar)
 
// =====================================================================
//                          ORDER BLOCKS
// =====================================================================
findLastBearOffset(fromBar) =>
    int result = na
    if not na(fromBar)
        int dist = math.min(bar_index - fromBar, 100)
        for i = 0 to dist
            if close[i] < open[i]
                result := i
                break
    result
 
findLastBullOffset(fromBar) =>
    int result = na
    if not na(fromBar)
        int dist = math.min(bar_index - fromBar, 100)
        for i = 0 to dist
            if close[i] > open[i]
                result := i
                break
    result
 
if showOB and (isBullBOS or isBullCHoCH)
    off = findLastBearOffset(lastPLbar)
    if not na(off)
        bool strongOB = not obVolFilter or (not na(volSma[off]) and volume[off] > volSma[off] * volMult)
        b = box.new(bar_index - off, high[off], bar_index, low[off],
          bgcolor      = bullOBCol,
          border_color = obBorder ? color.new(bullCol, strongOB ? 0 : 50) : color.new(color.black, 100),
          border_width = strongOB ? 2 : 1,
          extend       = extend.right)
        array.push(bullOBs, b)
        if aNewOB
            alert("New bullish Order Block" + (strongOB ? " (strong volume)" : ""), alert.freq_once_per_bar)
        while array.size(bullOBs) > obMaxCount
            box.delete(array.shift(bullOBs))
 
if showOB and (isBearBOS or isBearCHoCH)
    off = findLastBullOffset(lastPHbar)
    if not na(off)
        bool strongOB = not obVolFilter or (not na(volSma[off]) and volume[off] > volSma[off] * volMult)
        b = box.new(bar_index - off, high[off], bar_index, low[off],
          bgcolor      = bearOBCol,
          border_color = obBorder ? color.new(bearCol, strongOB ? 0 : 50) : color.new(color.black, 100),
          border_width = strongOB ? 2 : 1,
          extend       = extend.right)
        array.push(bearOBs, b)
        if aNewOB
            alert("New bearish Order Block" + (strongOB ? " (strong volume)" : ""), alert.freq_once_per_bar)
        while array.size(bearOBs) > obMaxCount
            box.delete(array.shift(bearOBs))
 
// =========== Helper: convert OB -> Breaker ===========
convertToBreaker(b, isBullToBear) =>
    if showBR
        topB  = box.get_top(b)
        botB  = box.get_bottom(b)
        leftB = box.get_left(b)
        col   = isBullToBear ? bearBRCol : bullBRCol
        edge  = isBullToBear ? bearCol   : bullCol
        br = box.new(leftB, topB, bar_index, botB,
          bgcolor      = col,
          border_color = color.new(edge, 30),
          border_style = line.style_dashed,
          border_width = 1,
          extend       = extend.right)
        if isBullToBear
            array.push(bearBreakers, br)
            while array.size(bearBreakers) > brMaxCount
                box.delete(array.shift(bearBreakers))
        else
            array.push(bullBreakers, br)
            while array.size(bullBreakers) > brMaxCount
                box.delete(array.shift(bullBreakers))
        if aBR
            alert("New " + (isBullToBear ? "bearish" : "bullish") + " Breaker Block", alert.freq_once_per_bar)
 
// OB touch flags on the current bar (for confluence signals)
bool tapBullOB = false
bool tapBearOB = false
 
// Touches and mitigation of bullish OBs -> a broken bullish OB = bearish breaker
if array.size(bullOBs) > 0
    for i = array.size(bullOBs) - 1 to 0
        b   = array.get(bullOBs, i)
        top = box.get_top(b)
        bot = box.get_bottom(b)
        bool touched = low <= top and high >= bot and not (low[1] <= top and high[1] >= bot)
        if touched
            tapBullOB := true
            if aTouch
                alert("Bullish Order Block touch", alert.freq_once_per_bar)
        bool kill = obMitigation == "Wick" ? low < bot : close < bot
        if kill
            array.remove(bullOBs, i)
            convertToBreaker(b, true)
            box.delete(b)
 
// Touches and mitigation of bearish OBs -> a broken bearish OB = bullish breaker
if array.size(bearOBs) > 0
    for i = array.size(bearOBs) - 1 to 0
        b   = array.get(bearOBs, i)
        top = box.get_top(b)
        bot = box.get_bottom(b)
        bool touched = low <= top and high >= bot and not (low[1] <= top and high[1] >= bot)
        if touched
            tapBearOB := true
            if aTouch
                alert("Bearish Order Block touch", alert.freq_once_per_bar)
        bool kill = obMitigation == "Wick" ? high > top : close > top
        if kill
            array.remove(bearOBs, i)
            convertToBreaker(b, false)
            box.delete(b)
 
// Breaker mitigation (when the breaker itself is broken - delete it)
if array.size(bullBreakers) > 0
    for i = array.size(bullBreakers) - 1 to 0
        b   = array.get(bullBreakers, i)
        bot = box.get_bottom(b)
        if close < bot
            array.remove(bullBreakers, i)
            box.delete(b)
 
if array.size(bearBreakers) > 0
    for i = array.size(bearBreakers) - 1 to 0
        b   = array.get(bearBreakers, i)
        top = box.get_top(b)
        if close > top
            array.remove(bearBreakers, i)
            box.delete(b)
 
// =====================================================================
//                          FAIR VALUE GAPS
// =====================================================================
if showFVG and bar_index > 2
    // Body of the central gap candle (for the displacement filter)
    float midBody = math.abs(close[1] - open[1])
    bool  dispOK  = not fvgDisp or midBody >= atr14 * fvgDispMult
    // Bullish FVG: low[0] > high[2]
    if low > high[2]
        float gap = low - high[2]
        if gap >= atr14 * fvgMinAtr and dispOK
            b = box.new(bar_index - 2, low, bar_index, high[2],
              bgcolor     = bullFVGCol,
              border_color= color.new(color.black, 100),
              extend      = extend.right)
            array.push(bullFVGs, b)
            if aFVG
                alert("New bullish FVG", alert.freq_once_per_bar)
            while array.size(bullFVGs) > fvgMaxCount
                box.delete(array.shift(bullFVGs))
    // Bearish FVG: high[0] < low[2]
    if high < low[2]
        float gap = low[2] - high
        if gap >= atr14 * fvgMinAtr and dispOK
            b = box.new(bar_index - 2, low[2], bar_index, high,
              bgcolor     = bearFVGCol,
              border_color= color.new(color.black, 100),
              extend      = extend.right)
            array.push(bearFVGs, b)
            if aFVG
                alert("New bearish FVG", alert.freq_once_per_bar)
            while array.size(bearFVGs) > fvgMaxCount
                box.delete(array.shift(bearFVGs))
 
if array.size(bullFVGs) > 0
    for i = array.size(bullFVGs) - 1 to 0
        b   = array.get(bullFVGs, i)
        bot = box.get_bottom(b)
        if low <= bot
            array.remove(bullFVGs, i)
            box.delete(b)
 
if array.size(bearFVGs) > 0
    for i = array.size(bearFVGs) - 1 to 0
        b   = array.get(bearFVGs, i)
        top = box.get_top(b)
        if high >= top
            array.remove(bearFVGs, i)
            box.delete(b)
 
// =====================================================================
//                          PREMIUM / DISCOUNT ZONES
// =====================================================================
// Take the last swing high and swing low, build Fib zones.
float pdMid = na
if not na(lastPH) and not na(lastPL)
    pdMid := math.avg(lastPH, lastPL)
 
if showPD and not na(lastPH) and not na(lastPL)
    float pdHi = lastPH
    float pdLo = lastPL
    int   lft  = math.max(lastPHbar, lastPLbar)
    int   rgt  = bar_index + pdExtend
    float eq   = pdMid
 
    // Create once, then update in place
    if na(pdBoxPrem)
        pdBoxPrem := box.new(lft, pdHi, rgt, eq,
          bgcolor      = premiumCol,
          border_color = color.new(color.black, 100))
        pdBoxDisc := box.new(lft, eq, rgt, pdLo,
          bgcolor      = discountCol,
          border_color = color.new(color.black, 100))
        pdLineEq  := line.new(lft, eq, rgt, eq, color=eqCol2, style=line.style_dashed, width=1)
        pdLineTop := line.new(lft, pdHi, rgt, pdHi, color=color.new(bearCol, 40), style=line.style_solid)
        pdLineBot := line.new(lft, pdLo, rgt, pdLo, color=color.new(bullCol, 40), style=line.style_solid)
    else
        box.set_lefttop(pdBoxPrem, lft, pdHi)
        box.set_rightbottom(pdBoxPrem, rgt, eq)
        box.set_bgcolor(pdBoxPrem, premiumCol)
        box.set_lefttop(pdBoxDisc, lft, eq)
        box.set_rightbottom(pdBoxDisc, rgt, pdLo)

// ---------- Smart visibility of Order Block zones ----------
gSV       = "Smart visibility"
smartView = input.bool(false, "Smart visibility: show only nearest OBs", group=gSV, tooltip="On the last bar, keep only the order blocks closest to price visible and hide the rest, to declutter. Affects display only, not detection.")
svCount   = input.int(5, "Max nearest OB zones to show", minval=1, maxval=20, group=gSV)
 
f_obVisible(b, vis, isBull) =>
    if vis
        box.set_bgcolor(b, isBull ? bullOBCol : bearOBCol)
        box.set_border_color(b, color.new(isBull ? bullCol : bearCol, 0))
    else
        box.set_bgcolor(b, color.new(color.gray, 100))
        box.set_border_color(b, color.new(color.gray, 100))
 
var bool svWasOn = false
if smartView and barstate.islast
    allDist = array.new<float>()
    if array.size(bullOBs) > 0
        for i = 0 to array.size(bullOBs) - 1
            bb = array.get(bullOBs, i)
            array.push(allDist, math.abs(close - math.avg(box.get_top(bb), box.get_bottom(bb))))
    if array.size(bearOBs) > 0
        for i = 0 to array.size(bearOBs) - 1
            bb = array.get(bearOBs, i)
            array.push(allDist, math.abs(close - math.avg(box.get_top(bb), box.get_bottom(bb))))
    float cutoff = na
    if array.size(allDist) > 0
        array.sort(allDist, order.ascending)
        int k = int(math.min(svCount, array.size(allDist))) - 1
        cutoff := array.get(allDist, k)
    if array.size(bullOBs) > 0
        for i = 0 to array.size(bullOBs) - 1
            bb = array.get(bullOBs, i)
            d  = math.abs(close - math.avg(box.get_top(bb), box.get_bottom(bb)))
            f_obVisible(bb, na(cutoff) or d <= cutoff, true)
    if array.size(bearOBs) > 0
        for i = 0 to array.size(bearOBs) - 1
            bb = array.get(bearOBs, i)
            d  = math.abs(close - math.avg(box.get_top(bb), box.get_bottom(bb)))
            f_obVisible(bb, na(cutoff) or d <= cutoff, false)
    svWasOn := true
else if (not smartView) and svWasOn and barstate.islast
    if array.size(bullOBs) > 0
        for i = 0 to array.size(bullOBs) - 1
            f_obVisible(array.get(bullOBs, i), true, true)
    if array.size(bearOBs) > 0
        for i = 0 to array.size(bearOBs) - 1
            f_obVisible(array.get(bearOBs, i), true, false)
    svWasOn := false
