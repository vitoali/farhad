safe_denominator(value) =>
    math.max(math.abs(nz(value, syminfo.mintick)), syminfo.mintick)

normalize_price_distance(distance, reference_price) =>
    distance / safe_denominator(reference_price)

restore_price_distance(normalized_distance, reference_price) =>
    normalized_distance * safe_denominator(reference_price)

clamp_score(value) =>
    math.max(0.0, math.min(100.0, value))

//---------------------------------------------------------------------------------------------------------------------
// Advanced Signal Filtering System Functions
//---------------------------------------------------------------------------------------------------------------------
// 1. Adaptive Clustering Prevention
is_signal_clustered(current_time, current_price, current_confidence) =>
    bool clustered = false
    
    if enable_adaptive_clustering and array.size(signal_timestamps) > 0
        // Calculate adaptive time and price windows based on volatility
        // Use atr_val as fallback if atr_dynamic is not yet available
        current_atr = safe_denominator(na(atr_dynamic) ? atr_val : atr_dynamic)
        current_avg_atr = safe_denominator(na(avg_atr) ? atr_val : avg_atr)
        volatility_factor = current_atr / current_avg_atr
        adaptive_time_window = clustering_time_window * math.max(0.5, math.min(2.0, volatility_factor))
        adaptive_price_window = clustering_price_window * current_atr * volatility_factor
        
        // Check for clustering using exponential decay
        int cluster_count = 0
        for i = 0 to array.size(signal_timestamps) - 1
            time_diff = current_time - array.get(signal_timestamps, i)
            price_diff = math.abs(current_price - array.get(signal_prices, i))
            confidence_diff = math.abs(current_confidence - array.get(signal_confidences, i))
            
            // Apply exponential decay to thresholds
            decay_factor = math.pow(clustering_decay_factor, time_diff / adaptive_time_window)
            decayed_time_threshold = adaptive_time_window * decay_factor
            decayed_price_threshold = adaptive_price_window * decay_factor
            
            if time_diff <= decayed_time_threshold and price_diff <= decayed_price_threshold
                cluster_count += 1
        
        // Check if this creates too many clustered signals
        if cluster_count >= max_clustered_signals
            clustered := true
    
    clustered

update_signal_history(current_time, current_price, current_confidence) =>
    if enable_adaptive_clustering
        // Add new signal to history
        array.push(signal_timestamps, current_time)
        array.push(signal_prices, current_price)
        array.push(signal_confidences, current_confidence)
        
        // Remove old signals beyond lookback period
        max_lookback = clustering_time_window * 3
        while array.size(signal_timestamps) > 0 and (current_time - array.get(signal_timestamps, 0)) > max_lookback
            array.shift(signal_timestamps)
            array.shift(signal_prices)
            array.shift(signal_confidences)

// 2. Regime-Aware Threshold Adjustment with Bayesian Learning
var float current_regime = 2.0  // 1=trending, 2=ranging

detect_market_regime() =>
    float regime = 2.0  // Default to ranging
    
    if enable_regime_adaptation
        // Calculate trend strength using multiple indicators
        current_atr = safe_denominator(na(atr_dynamic) ? atr_val : atr_dynamic)
        current_avg_atr = safe_denominator(na(avg_atr) ? atr_val : avg_atr)
        current_avg_volume = safe_denominator(na(avg_volume) ? volume : avg_volume)
        
        price_trend = ta.change(close, regime_lookback)
        volatility_trend = ta.change(current_atr, regime_lookback / 2)
        volume_trend = ta.change(volume, regime_lookback / 4)
        
        // Normalize trends by ATR
        normalized_price_trend = price_trend / current_atr
        normalized_volatility_trend = volatility_trend / current_avg_atr
        normalized_volume_trend = volume_trend / current_avg_volume
        
        // Regime classification
        trend_strength = math.abs(normalized_price_trend) + math.abs(normalized_volatility_trend) + math.abs(normalized_volume_trend)
        
        if trend_strength > volatility_regime_threshold
            regime := 1.0  // Trending
        else
            regime := 2.0  // Ranging
        

    
    regime

update_bayesian_thresholds(signal_confidence, signal_success) =>
    if enable_regime_adaptation
        array.push(historical_performance, signal_success ? 1.0 : 0.0)
        if array.size(historical_performance) > regime_lookback
            array.shift(historical_performance)

        // Bayesian updating of thresholds based on performance
        learning_rate = bayesian_learning_rate
        current_threshold = array.size(regime_thresholds) > 0 ? array.get(regime_thresholds, array.size(regime_thresholds) - 1) : min_signal_confidence
        performance_bias = signal_success ? signal_confidence : 100.0 - signal_confidence
        
        // Update threshold based on signal success
        float new_threshold = 0.0
        if signal_success
            // Successful signal: increase threshold slightly
            new_threshold := current_threshold + (learning_rate * (performance_bias - current_threshold) * 0.1)
        else
            // Failed signal: decrease threshold
            new_threshold := current_threshold - (learning_rate * performance_bias * 0.2)
        
        // Apply bounds
        new_threshold := math.max(50.0, math.min(95.0, new_threshold))
        
        // Store updated threshold
        array.push(regime_thresholds, new_threshold)
        
        // Keep only recent history
        if array.size(regime_thresholds) > regime_lookback
            array.shift(regime_thresholds)

calculate_adaptive_confidence_threshold() =>
    float threshold = min_signal_confidence
    
    if enable_regime_adaptation
        // Adjust threshold based on market regime and historical performance
        regime_factor = current_regime == 1 ? 1.2 : 0.9  // Higher threshold for trending markets
        
        if array.size(regime_thresholds) > 0
            // Calculate confidence interval
            recent_thresholds = array.slice(regime_thresholds, math.max(0, array.size(regime_thresholds) - 20))
            mean_threshold = array.avg(recent_thresholds)
            std_threshold = array.stdev(recent_thresholds)
            
            // Apply confidence interval (using normal distribution approximation)
            // For 95% confidence level, use 1.96 standard deviations
            confidence_multiplier = confidence_interval_level >= 0.95 ? 1.96 : 
                                   confidence_interval_level >= 0.90 ? 1.65 : 
                                   confidence_interval_level >= 0.80 ? 1.28 : 1.0
            confidence_interval = std_threshold * confidence_multiplier
            threshold := mean_threshold + confidence_interval
        
        // Apply regime adjustment
        threshold := threshold * regime_factor
        threshold := math.max(50.0, math.min(95.0, threshold))
    
    threshold

        if final_confidence < adaptive_threshold
            valid := false
    
    // Update signal history if valid
    if valid
        update_signal_history(bar_index, close, final_confidence)
    
    [valid, final_confidence]

// Kernel Functions

rbf_kernel(x1, x2) => 
    diff = (x1 - x2) / window
    math.exp(-math.pow(diff, 2) / (2.0 * math.pow(rbf_lengthscale, 2)))

linear_kernel(x1, x2) => 
    (x1 * x2) / (window * window)

polynomial_kernel(x1, x2) => 
    norm_x1 = x1 / window
    norm_x2 = x2 / window
    base = poly_gamma * norm_x1 * norm_x2 + poly_coef0
    math.pow(math.max(0.001, base), poly_degree)

sigmoid_kernel(x1, x2) => 
    norm_x1 = x1 / window
    norm_x2 = x2 / window
    val = sigmoid_gamma * norm_x1 * norm_x2 + sigmoid_coef0
    val := math.max(-10, math.min(10, val))
    (math.exp(2 * val) - 1) / (math.exp(2 * val) + 1)

laplacian_kernel(x1, x2) => 
    normalized_distance = math.abs(x1 - x2) / window
    math.exp(-laplacian_gamma * normalized_distance)

matern_kernel(x1, x2) => 
    r = math.abs(x1 - x2) / (matern_length_scale * window)
    if matern_nu == 0.5
        math.exp(-r)
    else if matern_nu == 1.5
        (1 + math.sqrt(3) * r) * math.exp(-math.sqrt(3) * r)
    else
        (1 + math.sqrt(5) * r + (5 * r * r) / 3) * math.exp(-math.sqrt(5) * r)

periodic_kernel(x1, x2) => 
    normalized_distance = math.abs(x1 - x2) / window
    sin_term = math.sin(math.pi * normalized_distance * window / periodic_period)
    math.exp(-(periodic_length_scale * periodic_length_scale / 2) * sin_term * sin_term)

trigo_kernel(x1, x2) =>
    base_ds1 = x1 - x2
    normalized_base = base_ds1 / window
    synth_ds1 = math.sin(0.68 * math.pow(normalized_base, 2) + 0.79 * normalized_base + normalized_base) * math.cos(0.68 * math.pow(normalized_base, 2) + 0.79 * normalized_base + normalized_base)
    math.exp(-math.abs(synth_ds1))

compute_single_kernel(x1, x2, kernelType) =>
    switch kernelType
        "RBF" => rbf_kernel(x1, x2)
        "Linear" => linear_kernel(x1, x2)
        "Polynomial" => polynomial_kernel(x1, x2)
        "Sigmoid" => sigmoid_kernel(x1, x2)
        "Laplacian" => laplacian_kernel(x1, x2)
        "Matérn" => matern_kernel(x1, x2)
        "Periodic" => periodic_kernel(x1, x2)
        "Trigo" => trigo_kernel(x1, x2)
        => rbf_kernel(x1, x2)

safe_kernel_base(kernel_value) =>
    math.max(nz(kernel_value, 0.0), 0.0)

safe_multiplicative_kernel(k1, k2, w1, w2) =>
    math.pow(safe_kernel_base(k1), w1) * math.pow(safe_kernel_base(k2), w2)

compute_mlma_kernel(x1, x2) =>
    if enableMlmaMultiKernel
        k1 = compute_single_kernel(x1, x2, mlma_kernel1_type)
        k2 = compute_single_kernel(x1, x2, mlma_kernel2_type)
        total_weight = mlma_kernel1_weight + mlma_kernel2_weight
        w1 = total_weight > 0 ? mlma_kernel1_weight / total_weight : 0.5
        w2 = total_weight > 0 ? mlma_kernel2_weight / total_weight : 0.5
        if mlmaMultiKernelMethod == "Additive"
            w1 * k1 + w2 * k2
        else
            safe_multiplicative_kernel(k1, k2, w1, w2)
    else
        compute_single_kernel(x1, x2, mlma_kernel_type)

compute_ob_kernel(x1, x2) =>
    if enableObMultiKernel
        k1 = compute_single_kernel(x1, x2, ob_kernel1_type)
        k2 = compute_single_kernel(x1, x2, ob_kernel2_type)
        total_weight = ob_kernel1_weight + ob_kernel2_weight
        w1 = total_weight > 0 ? ob_kernel1_weight / total_weight : 0.5
        w2 = total_weight > 0 ? ob_kernel2_weight / total_weight : 0.5
        if obMultiKernelMethod == "Additive"
            w1 * k1 + w2 * k2
        else
            safe_multiplicative_kernel(k1, k2, w1, w2)
    else
        compute_single_kernel(x1, x2, ob_kernel_type)

build_mlma_kernel_matrix(X1, X2) =>
    rows = X1.size()
    cols = X2.size()
    K = matrix.new<float>(rows, cols, 0.0)
    for i = 0 to rows - 1
        x1 = X1.get(i)
        for j = 0 to cols - 1
            x2 = X2.get(j)
            k_val = compute_mlma_kernel(x1, x2)
            K.set(i, j, k_val)
    K

            k_val = compute_ob_kernel(x1, x2)
            K.set(i, j, k_val)
    K

gpr_predict(K_train, K_test) =>
    n = K_train.rows()
    I = matrix.new<float>(n, n, 0.0)
    for i = 0 to n - 1
        I.set(i, i, 1.0)
    noise_matrix = I.mult(sigma * sigma)
    K_noise = K_train.sum(noise_matrix)
    K_inv = K_noise.pinv()
    weights = K_test.transpose().mult(K_inv)
    weights.row(window + forecast - 1)

krr_predict(K_train, K_test) =>
    n = K_train.rows()
    I = matrix.new<float>(n, n, 0.0)
    for i = 0 to n - 1
        I.set(i, i, 1.0)
    reg_matrix = I.mult(lambda_krr)
    K_reg = K_train.sum(reg_matrix)
    K_inv = K_reg.pinv()
    weights = K_test.transpose().mult(K_inv)
    weights.row(window + forecast - 1)

kpcr_predict(K_train, K_test) =>
    n = K_train.rows()
    I = matrix.new<float>(n, n, 0.0)
    for i = 0 to n - 1
        I.set(i, i, 1.0)
    component_count = math.max(1, math.min(window, n_components))
    effective_kpcr_regularization = kpcr_regularization * (window / component_count)
    reg_matrix = I.mult(effective_kpcr_regularization)
    K_reg = K_train.sum(reg_matrix)
    K_inv = K_reg.pinv()
    weights = K_test.transpose().mult(K_inv)
    weights.row(window + forecast - 1)

svr_predict(K_train, K_test) =>
    n = K_train.rows()
    I = matrix.new<float>(n, n, 0.0)
    for i = 0 to n - 1
        I.set(i, i, 1.0)
    effective_svr_regularization = (1.0 / C_svr) + epsilon_svr
    reg_matrix = I.mult(effective_svr_regularization)
    K_reg = K_train.sum(reg_matrix)
    K_inv = K_reg.pinv()
    weights = K_test.transpose().mult(K_inv)
    weights.row(window + forecast - 1)

kernel_smoothing_predict(K_test) =>
    test_row = K_test.transpose().row(window + forecast - 1)
    total_weight = 0.0
    for i = 0 to test_row.size() - 1
        bandwidth_weight = math.exp(-math.pow((window + forecast - 1 - i) / safe_denominator(bandwidth), 2) / 2.0)
        total_weight += math.max(test_row.get(i), 0.0) * bandwidth_weight
    normalized_weights = array.new<float>(0)
    if total_weight > 0
        for i = 0 to test_row.size() - 1
            bandwidth_weight = math.exp(-math.pow((window + forecast - 1 - i) / safe_denominator(bandwidth), 2) / 2.0)
            normalized_weights.push((math.max(test_row.get(i), 0.0) * bandwidth_weight) / total_weight)
    else
        equal_weight = 1.0 / test_row.size()
        for i = 0 to test_row.size() - 1
            normalized_weights.push(equal_weight)
    normalized_weights

adaptive_rq_predict(rq_src, length, forecast_steps) =>
    float weighted_sum = 0.0
    float total_weight = 0.0
    current_atr_rq = safe_denominator(ta.atr(fib_atr_period))
    avg_atr_rq = safe_denominator(ta.sma(current_atr_rq, length))
    volatility_factor = rq_volatility_adapt ? math.max(0.5, math.min(2.0, current_atr_rq / avg_atr_rq)) : 1.0
    effective_bandwidth = math.max(1.0, rq_bandwidth * volatility_factor)
    target_index = length - 1 + forecast_steps
    
    for i = 0 to length - 1
        distance = math.abs(target_index - i)
        weight = math.pow(1.0 + (distance * distance) / (2.0 * rq_alpha * effective_bandwidth * effective_bandwidth), -rq_alpha)
        weighted_sum += weight * rq_src[length - 1 - i]
        total_weight += weight
    
    total_weight > 0 ? weighted_sum / total_weight : rq_src

calculate_exhaustion() =>
    volume_ratio = nz(volume) / safe_denominator(volume_ma)
    price_velocity = ta.change(close, 1)
    price_acceleration = ta.change(price_velocity, 1)
    volume_ma_falling = ta.falling(volume_ma, 3)
    exhaustion_highest = ta.highest(close, exhaustion_length * 2)[1]
    exhaustion_lowest = ta.lowest(close, exhaustion_length * 2)[1]
    
    bull_exhaustion = false
    bear_exhaustion = false
    
    if close > sma_20
        high_rsi = rsi_exhaustion > exhaustion_threshold
        declining_momentum = price_acceleration < 0 and price_velocity > 0
        volume_decline = volume_ratio < 0.8 and volume_ma_falling
        price_extension = close > exhaustion_highest
        bull_exhaustion := high_rsi and (declining_momentum or volume_decline) and price_extension
    
    if close < sma_20
        low_rsi = rsi_exhaustion < (100 - exhaustion_threshold)
        declining_selling = price_acceleration > 0 and price_velocity < 0
        volume_decline_bear = volume_ratio < 0.8 and volume_ma_falling
        price_extension_bear = close < exhaustion_lowest
        bear_exhaustion := low_rsi and (declining_selling or volume_decline_bear) and price_extension_bear
    
    bull_exhaustion ? 1 : bear_exhaustion ? -1 : 0

exhaustion_signal = calculate_exhaustion()

calculate_stop_loss(entry_price, is_bullish_position, swing_ref) =>
    sl_price = 0.0
    
    switch sl_method
        "ATR-based" =>
            sl_distance = atr_val * sl_atr_multiplier
            sl_price := is_bullish_position ? entry_price - sl_distance : entry_price + sl_distance
        
        "Swing-based" =>
            if not na(swing_ref)
                buffer = atr_val * sl_swing_buffer
                sl_price := is_bullish_position ? swing_ref - buffer : swing_ref + buffer
            else
                // Fallback to ATR method
                sl_distance = atr_val * sl_atr_multiplier
                sl_price := is_bullish_position ? entry_price - sl_distance : entry_price + sl_distance
        
        "Order Block" =>
            sl_price := na
        
        "Percentage" =>
            percentage_distance = entry_price * (sl_percentage / 100)
            sl_price := is_bullish_position ? entry_price - percentage_distance : entry_price + percentage_distance
    
    sl_price

detect_swing_high(lookback) =>
    float swing_h = na
    for i = 1 to lookback
        if high[i] > high[i-1] and high[i] > high[i+1]
            valid_swing = true
            if i < lookback - 1
                for j = 0 to math.min(5, lookback - i - 1)
                    if high[i] <= high[i+j+1]
                        valid_swing := false
                        break
            if valid_swing and (na(swing_h) or high[i] > swing_h)
                swing_h := high[i]
    swing_h

detect_swing_low(lookback) =>
    float swing_l = na
    for i = 1 to lookback
        if low[i] < low[i-1] and low[i] < low[i+1]
            valid_swing = true
            if i < lookback - 1
                for j = 0 to math.min(5, lookback - i - 1)
                    if low[i] >= low[i+j+1]
                        valid_swing := false
                        break
            if valid_swing and (na(swing_l) or low[i] < swing_l)
                swing_l := low[i]
    swing_l

    base_score = (trend_ok ? 35.0 : 0.0) + (slope_ok ? 25.0 : 0.0) + (price_ok ? 20.0 : 0.0) + (cloud_ok ? 15.0 : 0.0) + (near_ob ? 5.0 : 0.0)
    clamp_score(base_score)

calculate_volume_quality_score() =>
    volume_ratio = nz(volume) / safe_denominator(volume_ma)
    volume_score = volume_ratio >= 1.5 ? 100.0 : volume_ratio >= 1.1 ? 85.0 : volume_ratio >= 0.8 ? 60.0 : 35.0
    clamp_score(volume_score)

calculate_volatility_quality_score() =>
    volatility_ratio = safe_denominator(atr_dynamic) / safe_denominator(avg_atr)
    volatility_score = volatility_ratio < 0.55 ? 25.0 : volatility_ratio < 0.8 ? 60.0 : volatility_ratio <= 2.2 ? 90.0 : 55.0
    clamp_score(volatility_score)

calculate_momentum_quality_score(direction) =>
    candle_range = safe_denominator(high - low)
    candle_body_ratio = math.abs(close - open) / candle_range
    rsi_ok = direction == 1 ? rsi_14 > 50 and rsi_14 < 72 : rsi_14 < 50 and rsi_14 > 28
    candle_ok = direction == 1 ? close > open : close < open
    slope_ok = direction == 1 ? sma_slope > 0 : sma_slope < 0
    momentum_score = (rsi_ok ? 35.0 : 10.0) + (candle_ok ? 25.0 : 5.0) + (slope_ok ? 25.0 : 5.0) + (candle_body_ratio > 0.45 ? 15.0 : 5.0)
    clamp_score(momentum_score)

calculate_signal_quality_score(direction) =>
    mtf_score = calculate_mtf_alignment_score(direction)
    context_score = calculate_mlma_context_score(direction)
    volume_score = calculate_volume_quality_score()
    volatility_score = calculate_volatility_quality_score()
    momentum_score = calculate_momentum_quality_score(direction)
    ob_context_score = enable_ob_intelligence ? calculate_ob_context_quality_score(direction) : 70.0
    streak_penalty = math.min(35.0, quality_loss_streak * loss_streak_penalty)
    raw_score = (context_score * 0.27) + (mtf_score * 0.22) + (momentum_score * 0.18) + (volatility_score * 0.13) + (volume_score * 0.08) + (ob_context_score * 0.12) - streak_penalty
    clamp_score(raw_score)

is_no_trade_regime(direction) =>
    atr_base = safe_denominator(atr_dynamic)
    cloud_width_atr = math.abs(mlma_upper_final - mlma_lower_final) / atr_base
    mlma_slope_atr = math.abs(mlma_out_final - mlma_out_final[3]) / atr_base
    volatility_ratio = safe_denominator(atr_dynamic) / safe_denominator(avg_atr)
    too_choppy = strict_no_trade_filter and (isConsolidating or wedge_condition1 or (enable_dual_length and wedge_condition2) or (market_regime == 2 and mlma_slope_atr < 0.15))
    too_compressed = strict_no_trade_filter and volatility_ratio < 0.55
    trend_conflict = strict_no_trade_filter and not is_trend_aligned(direction, os_final)
    trapped_inside_cloud = strict_no_trade_filter and cloud_width_atr < 0.8 and close > mlma_lower_final and close < mlma_upper_final
    too_choppy or too_compressed or trend_conflict or trapped_inside_cloud

calcVolumePercentages(blocks, showLast) =>
    float totalVolume = 0.0
    percentages = array.new<float>()
    for i = 0 to math.min(showLast - 1, blocks.size() - 1)
        if blocks.size() > 0
            block = blocks.get(blocks.size() - 1 - i)
            totalVolume += block.vol
    for i = 0 to math.min(showLast - 1, blocks.size() - 1)
        if blocks.size() > 0
            block = blocks.get(blocks.size() - 1 - i)
            percentage = totalVolume > 0 ? (block.vol / totalVolume) * 100 : 0
            percentages.push(percentage)
    [totalVolume, percentages]

calculateStrengths(openn, highh, loww, closee, volumee) =>
    bullishStrength = (closee - loww) * volumee
    bearishStrength = (highh - closee) * volumee
    [bullishStrength, bearishStrength]

calculate_ob_quality(block_top, block_bottom, block_volume, bullish_strength, bearish_strength, block_type) =>
    block_size = math.max(block_top - block_bottom, syminfo.mintick)
    volume_ratio = block_volume / safe_denominator(avg_volume)
    volume_score = math.min(100.0, volume_ratio * 45.0)
    size_atr_ratio = block_size / safe_denominator(atr_dynamic)
    size_score = size_atr_ratio < 0.25 ? 35.0 : size_atr_ratio <= 1.8 ? 90.0 : math.max(35.0, 100.0 - ((size_atr_ratio - 1.8) * 22.0))
    total_strength = safe_denominator(bullish_strength + bearish_strength)
    pressure_score = block_type == "bullish" ? (bullish_strength / total_strength) * 100.0 : (bearish_strength / total_strength) * 100.0
    displacement_score = block_type == "bullish" ? math.min(100.0, math.max(0.0, (close - block_top) / safe_denominator(atr_dynamic)) * 55.0) : math.min(100.0, math.max(0.0, (block_bottom - close) / safe_denominator(atr_dynamic)) * 55.0)
    structure_score = block_type == "bullish" ? (close > ob_prior_structure_high ? 100.0 : close > block_top ? 65.0 : 25.0) : (close < ob_prior_structure_low ? 100.0 : close < block_bottom ? 65.0 : 25.0)
    trend_score = block_type == "bullish" ? (os_final == 1 ? 100.0 : 35.0) : (os_final == 0 ? 100.0 : 35.0)
    quality_score = (displacement_score * 0.25) + (trend_score * 0.20) + (volume_score * 0.18) + (pressure_score * 0.17) + (structure_score * 0.12) + (size_score * 0.08)
    clamp_score(quality_score)

isOverlapping(newBlock, existingArray) =>
    bool flag = false
    if existingArray.size() > 0
        for i = 0 to existingArray.size() - 1
            existingBlock = array.get(existingArray, i)
            if newBlock.bottom <= existingBlock.top and newBlock.top >= existingBlock.bottom
                flag := true
                break
    flag

    else if vol >= 1000
        str.tostring(vol / 1000, "#.#") + "K"
    else
        str.tostring(vol, "#")

    blockPercentage

// Order Block Strength Scoring System (0-100)
calculate_block_strength(block, block_age, block_type) =>
    float strength_score = 0.0
    
    // Volume component (40% weight)
    volume_percentile = block.vol / safe_denominator(avg_volume)
    volume_score = math.min(100, volume_percentile * 50)
    
    // Size component (25% weight)
    block_size = block.top - block.bottom
    size_atr_ratio = block_size / safe_denominator(atr_dynamic)
    size_score = math.min(100, size_atr_ratio * 50)
    
    // Age component (20% weight)
    age_score = math.max(0, 100 - (block_age * 2))
    
    // Momentum component (15% weight)
    total_strength = safe_denominator(block.bullishStr + block.bearishStr)
    momentum_score = block_type == "bullish" ? (block.bullishStr / total_strength) * 100 : (block.bearishStr / total_strength) * 100
    
    // Weighted calculation
    strength_score := (volume_score * 0.4) + (size_score * 0.25) + (age_score * 0.2) + (momentum_score * 0.15)
    strength_score

volume_confirmation_score(block_type) =>
    float score = 0.0
    
    // Volume increase requirement
    volume_increase = volume > avg_volume * 1.2
    
    // Volume trend analysis
    volume_trend = ta.rising(volume, 3)
    
    // Volume-price relationship
    price_volume_aligned = block_type == "bullish" ? 
                          (close > close[1] and volume > volume[1]) or (close < close[1] and volume < volume[1]) :
                          (close < close[1] and volume > volume[1]) or (close > close[1] and volume < volume[1])
    
    if volume_increase and volume_trend
        score := 80.0
    else if volume_increase
        score := 60.0
    else if price_volume_aligned
        score := 40.0
    else
        score := 20.0
    
    score

// Enhanced MLMA Trend Alignment Scoring
mlma_trend_score(block_type) =>
    float score = 0.0
    
    // Check if MLMA trend aligns with retest direction
    trend_aligned = block_type == "bullish" ? os_final == 1 : os_final == 0
    
    if trend_aligned
        // Perfect trend alignment
        score := 100.0
    else
        // Counter-trend retest - lower score but still possible
        score := 30.0
    
    // Bonus for MLMA cloud context
    if block_type == "bullish"
        if close >= mlma_lower_final and close <= mlma_upper_final
            score := score + 10.0  // Within MLMA cloud
    else
        if close >= mlma_lower_final and close <= mlma_upper_final
            score := score + 10.0  // Within MLMA cloud
    
    math.min(100.0, score)

// Volatility Regime Scoring
volatility_regime_score() =>
    float score = 0.0
    
    // Calculate volatility ratio
    volatility_ratio = safe_denominator(atr_dynamic) / safe_denominator(avg_atr)
    
    // Score based on volatility regime
    if volatility_ratio > 1.5
        // High volatility - good for momentum trades
        score := 80.0
    else if volatility_ratio < 0.5
        // Low volatility - good for range trades
        score := 60.0
    else
        // Normal volatility - balanced
        score := 70.0
    
    score

has_clean_retest_departure(block, block_type, block_age) =>
    bool moved_away = false
    bars_to_scan = math.min(retestLookbackBars, block_age)
    if block_age >= retestMinBars and bars_to_scan > 1
        for j = 1 to bars_to_scan
            if j <= bar_index
                if block_type == "bullish"
                    if high[j] >= block.top + atr_dynamic * retestPriceMovement and close[j] > block.top
                        moved_away := true
                        break
                else
                    if low[j] <= block.bottom - atr_dynamic * retestPriceMovement and close[j] < block.bottom
                        moved_away := true
                        break
    moved_away

// Enhanced Retest Confidence Calculation
calculate_enhanced_retest_confidence(block, block_type, block_index) =>
    float confidence_score = 0.0
    
    // Calculate individual component scores
    volume_score = volume_confirmation_score(block_type)
    mlma_score = mlma_trend_score(block_type)
    volatility_score = volatility_regime_score()
    block_age = bar_index - block.creationIndex
    strength_score = enable_ob_intelligence ? block.quality : calculate_block_strength(block, block_age, block_type)
    
    // Apply weights based on retest mode
    if retestMode == "Volume-Based"
        // Traditional volume-based scoring
        confidence_score := (volume_score * 0.4) + (strength_score * 0.6)
    else if retestMode == "Trend-Aligned"
        // MLMA and volatility focused
        confidence_score := (mlma_score * retestMlmaWeight) + 
                           (volatility_score * retestVolatilityWeight) + 
                           (strength_score * retestStrengthWeight) + 
                           (volume_score * retestVolumeWeight)
    else  // Hybrid mode
        // Balanced approach
        confidence_score := (mlma_score * 0.25) + 
                           (volatility_score * 0.20) + 
                           (strength_score * 0.30) + 
                           (volume_score * 0.25)
    
    retest_penalty = enable_ob_intelligence ? math.min(25.0, block.retests * 8.0) : 0.0
    math.max(0.0, math.min(100.0, confidence_score - retest_penalty))

// Intelligent Retest Detection Function
is_intelligent_retest(block, block_index, block_type, block_being_tested_array, last_touch_bar_array) =>
    bool valid_retest = false
    float confidence_score = 0.0
    
    tolerance = atr_dynamic * retestTolerance
    block_age = bar_index - block.creationIndex
    touching_block = high >= block.bottom - tolerance and low <= block.top + tolerance
    block_not_broken = block_type == "bullish" ? close >= block.bottom - tolerance : close <= block.top + tolerance
    
    if touching_block and block_not_broken
        // Check if this is a new retest (not currently being tested)
        currently_testing = array.get(block_being_tested_array, block_index)
        last_touch = array.get(last_touch_bar_array, block_index)
        
        clean_departure = has_clean_retest_departure(block, block_type, block_age)
        enough_bars_since_touch = last_touch == 0 or bar_index - last_touch > retestMinBars
        
        if currently_testing
            // Already testing this block - don't show another signal
            valid_retest := false
        
        // Additional filters for quality based on retest mode
        volume_ok = volume > avg_volume * retestVolumeThreshold  // User-controlled volume threshold
        strength_ok = enable_ob_intelligence ? block.quality > retestStrengthThreshold : calculate_block_strength(block, block_age, block_type) > retestStrengthThreshold
        retest_count_ok = block.retests < retestMaxCount  // User-controlled max retests per block
        block_range = safe_denominator(block.top - block.bottom)
        reaction_ok = block_type == "bullish" ? close >= block.bottom and close >= low + block_range * 0.15 : close <= block.top and close <= high - block_range * 0.15
        
        // Trend alignment filter for Trend-Aligned mode
        trend_ok = true
        if retestMode == "Trend-Aligned" and retestTrendAlignment
            trend_ok = block_type == "bullish" ? os_final == 1 : os_final == 0
        
        // Volatility regime filter
        volatility_ok = true
        if retestMode == "Trend-Aligned" and retestVolatilityRegime
            volatility_ratio = safe_denominator(atr_dynamic) / safe_denominator(avg_atr)
            volatility_ok = volatility_ratio > 0.3 and volatility_ratio < 3.0  // Reasonable volatility range
        
        if (not currently_testing and clean_departure and enough_bars_since_touch) and volume_ok and strength_ok and retest_count_ok and trend_ok and volatility_ok and reaction_ok
            // Calculate enhanced confidence score
            confidence_score := calculate_enhanced_retest_confidence(block, block_type, block_index)
            
            // Apply advanced signal filtering
            [advanced_valid, advanced_confidence] = is_advanced_signal_valid(block, block_type, confidence_score)
            
            if advanced_valid
                valid_retest := true
                confidence_score := advanced_confidence
                
                // Update testing state - mark as currently being tested
                array.set(block_being_tested_array, block_index, true)
                array.set(last_touch_bar_array, block_index, bar_index)
    
    [valid_retest, confidence_score]

    if not enableProximityEffect
        cloudIntensity
    else
        bandDistance = math.abs(bandValue - mlmaValue)
        priceDistance = math.abs(price - mlmaValue)

            proximityFactor = 1 - (priceDistance / bandDistance)
            proximityFactor := math.max(0.2, math.min(1.0, proximityFactor))
            cloudIntensity * proximityFactor
