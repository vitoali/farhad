f_clamp(float v, float lo, float hi) => math.max(lo, math.min(hi, v))

f_fmt(float v) =>
    float a = math.abs(v)
    string s = v < 0 ? "-" : ""
    a >= 1e9 ? s + str.tostring(a / 1e9, "#.##") + "B" : a >= 1e6 ? s + str.tostring(a / 1e6, "#.##") + "M" : a >= 1e3 ? s + str.tostring(a / 1e3, "#.##") + "K" : s + str.tostring(a, "#")

f_txt(string s) =>
    switch s
        "Tiny"   => size.tiny
        "Small"  => size.small
        "Normal" => size.normal
        "Large"  => size.large
        => size.small

f_dpos(string p) =>
    switch p
        "Top Left"     => position.top_left
        "Bottom Right" => position.bottom_right
        "Bottom Left"  => position.bottom_left
        "Middle Right" => position.middle_right
        => position.top_right

f_stars(float s) => s >= 8 ? "★★★★★" : s >= 6.5 ? "★★★★" : s >= 5 ? "★★★" : s >= 3.5 ? "★★" : s >= 2 ? "★" : "·"
f_tier(float s)  => s >= 8 ? "PEAK" : s >= 6.5 ? "HIGH" : s >= 5 ? "FAIR" : s >= 3.5 ? "SOFT" : "WEAK"

f_tflabel(string tf) =>
    int s = timeframe.in_seconds(tf)
    s % 86400 == 0 ? "D" + str.tostring(s / 86400) : s >= 3600 and s % 3600 == 0 ? "H" + str.tostring(s / 3600) : s >= 60 ? "M" + str.tostring(s / 60) : "S" + str.tostring(s)

f_sess() =>
    int m = hour(time, "GMT") * 60 + minute(time, "GMT")
    m >= 480 and m < 780 ? "London" : m >= 780 and m < 1320 ? "NY" : m >= 0 and m < 540 ? "Asian" : "Other"

f_buy(float h, float l, float c, float v) => (h - l) > 0 ? v * (c - l) / (h - l) : v * 0.5
f_abs(float o, float h, float l, float c, float v) =>
    float rng = h - l
    rng > 0 ? v * ((h - math.max(o, c)) + (math.min(o, c) - l)) / rng * 0.5 : 0.0

f_strength(bool isBull, float gap, float atrv, float fVol, float fDlt, float fAbs, float avgV, float midBody, float midRng) =>
    float sizeS  = f_clamp(gap / math.max(atrv * 1.5, 1e-9), 0.0, 1.0)
    float dir    = isBull ? fDlt : -fDlt
    float alignS = fVol > 0 ? f_clamp(dir / fVol, 0.0, 1.0) : 0.0
    float rvolS  = avgV > 0 ? f_clamp(fVol / (avgV * 3.0) / 2.0, 0.0, 1.0) : 0.0
    float domS   = f_clamp(math.abs(fDlt) / math.max(fVol, 1.0), 0.0, 1.0)
    float dispS  = midRng > 0 ? f_clamp(math.abs(midBody) / midRng, 0.0, 1.0) : 0.0
    float absS   = f_clamp(fAbs / math.max(fVol, 1.0) * 2.0, 0.0, 1.0)
    float wSum   = math.max(i_wSize + i_wAlign + i_wRvol + i_wDom + i_wDisp + i_wAbs, 1e-6)
    float sc = (sizeS * i_wSize + alignS * i_wAlign + rvolS * i_wRvol + domS * i_wDom + dispS * i_wDisp + absS * i_wAbs) / wSum * 10.0
    f_clamp(sc, 0.0, 10.0)

detect_bull() =>
    var float a_top = na
    var float a_bot = na
    var float a_sc = 0.0
    var int a_t = na
    var float a_buy = 0.0
    var float a_vol = 0.0
    var float a_dlt = 0.0
    var float a_abs = 0.0
    var float a_rv = 0.0
    float atrv = ta.atr(i_atrLen)[1]
    float avgV = ta.sma(volume, i_rvolLen)[1]
    if low[1] > high[3]
        float gtop = low[1]
        float gbot = high[3]
        float gap = gtop - gbot
        float minGap = i_useATR ? atrv * i_atrMult : (i_minGapPct > 0 ? close[1] * i_minGapPct / 100.0 : 0.0)
        if gap >= minGap
            float b1 = f_buy(high[1], low[1], close[1], volume[1])
            float b2 = f_buy(high[2], low[2], close[2], volume[2])
            float b3 = f_buy(high[3], low[3], close[3], volume[3])
            a_buy := b1 + b2 + b3
            a_vol := volume[1] + volume[2] + volume[3]
            a_dlt := 2.0 * a_buy - a_vol
            a_abs := f_abs(open[1], high[1], low[1], close[1], volume[1]) + f_abs(open[2], high[2], low[2], close[2], volume[2]) + f_abs(open[3], high[3], low[3], close[3], volume[3])
            a_rv  := avgV > 0 ? a_vol / (avgV * 3.0) * 100.0 : 100.0
            a_top := gtop
            a_bot := gbot
            a_sc  := f_strength(true, gap, atrv, a_vol, a_dlt, a_abs, avgV, close[2] - open[2], high[2] - low[2])
            a_t   := time[1]
    if not na(a_bot) and close[1] < a_bot
        a_top := na
        a_bot := na
        a_sc  := 0.0
        a_t   := na
    [a_top, a_bot, a_sc, a_t, a_buy, a_vol, a_dlt, a_abs, a_rv]

        float m = (rawTop + rawBot) / 2.0
        topP := m + h / 2.0
        botP := m - h / 2.0
    float bPct = totv > 0 ? buyv / totv * 100.0 : 50.0
    color zc = isBull ? i_bullZone : i_bearZone
    int   ztr = int(f_clamp(90.0 - sc * 4.0, 55.0, 90.0))

f_spawn(bool isBull, int tfIdx, float top, float bot, float sc, float buyv, float totv, float dlt, float ab, float rv) =>
    if not na(top) and not na(bot) and sc >= i_minStr
        zones.push(f_build(isBull, tfIdx == 1, tfIdx == 2, tfIdx == 3, top, bot, sc, buyv, totv, dlt, ab, rv))

f_absorbInto(OFZone a, OFZone b) =>
    a.hasTf1 := a.hasTf1 or b.hasTf1
    a.hasTf2 := a.hasTf2 or b.hasTf2
    a.hasTf3 := a.hasTf3 or b.hasTf3
    float wA = math.max(a.totVol, 1.0)
    float wB = math.max(b.totVol, 1.0)
    float wT = wA + wB
    float aMid = math.avg(a.topP, a.botP)
    float bMid = math.avg(b.topP, b.botP)
    float nMid = (aMid * wA + bMid * wB) / wT
    a.buyVol += b.buyVol
    a.totVol += b.totVol
    a.absorb += b.absorb
    a.delta  := 2.0 * a.buyVol - a.totVol
    a.bullPct := a.totVol > 0 ? a.buyVol / a.totVol * 100.0 : 50.0
    a.bearPct := 100.0 - a.bullPct
    a.relVol  := (a.relVol * wA + b.relVol * wB) / wT
    float h = i_norm ? f_normH() : (a.topP - a.botP)
    a.topP := nMid + h / 2.0
    a.botP := nMid - h / 2.0
    int cc = (a.hasTf1 ? 1 : 0) + (a.hasTf2 ? 1 : 0) + (a.hasTf3 ? 1 : 0)
    a.scoreBase := math.max(a.scoreBase, b.scoreBase)
    a.score := f_clamp(a.scoreBase + i_confBonus * (cc - 1), 0.0, 10.0)
    a.tier := f_tier(a.score)
    b.body.delete()
    b.c1.delete()
    b.c2.delete()
    b.c3.delete()
    b.c4.delete()
    b.barBe.delete()
    b.barBu.delete()
    b.stat.delete()
    b.tag.delete()

f_consolidate() =>
    bool didMerge = false
    int n = zones.size()
    if n > 1
        for i = 0 to n - 2
            if didMerge
                break
            for j = i + 1 to n - 1
                OFZone A = zones.get(i)
                OFZone B = zones.get(j)
                if not A.mit and not B.mit and A.isBull == B.isBull
                    float tol = atrChart * i_mergeATR
                    if A.botP <= B.topP + tol and B.botP <= A.topP + tol
                        f_absorbInto(A, B)
                        zones.remove(j)
                        didMerge := true
                        break
    didMerge

if barstate.isconfirmed
    if i_bull
        if i_en1 and not na(b1x) and (na(lb1) or b1x != lb1)
            f_spawn(true, 1, b1t, b1b, b1s, b1by, b1v, b1d, b1a, b1r)

        if i_en2 and not na(b2x) and (na(lb2) or b2x != lb2)

        if i_en3 and not na(b3x) and (na(lb3) or b3x != lb3)
