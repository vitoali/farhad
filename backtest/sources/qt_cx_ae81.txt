// Constants
dur_hour  = 60 * 60 * 1000
dur_min   = 60 * 1000
msDay     = 24 * 60 * 60 * 1000
msWeek    = 7 * 24 * 60 * 60 * 1000

dur_day   = mdl == mdl1 ? free_day * dur_min : mdl == mdl2 ? 6 * dur_hour :  270 * dur_min
X = mdl == mdl1 ? X_ : mdl == mdl2 ? 4  : 3  
dur_sub   = dur_day / X
dur_micro = dur_sub / X
dur_mini  = dur_micro / X

start_hour = mdl == mdl1 ? start_hour_ : mdl == mdl2 ? 18 : mdl == mdl3 ? 2 : 2 
start_min  = mdl == mdl1 ? start_min_  : mdl == mdl2 ? 00 : mdl == mdl3 ? 30 : 00
// Time components
yy  = year(time, i_tz)
mo  = month(time, i_tz)
dom = dayofmonth(time, i_tz)
dow = dayofweek(time,  i_tz)

// Day start (timezone-aware)
dayStartToday    = timestamp(  i_tz, yy, mo, dom, start_hour, start_min)
dayStart         = time >= dayStartToday ? dayStartToday : dayStartToday - msDay
var extdaystart = time

extimestart    = timestamp(i_tz, yy, mo, dom, 16, 00)
extimefinish   = timestamp(i_tz, yy, mo, dom, 02, 30)


if timeframe.change("1D")
    extdaystart := time

extime     = time > extimestart or ( time >= extdaystart and time < extimefinish )

yy_c  = year(dayStart,   i_tz)
mo_c  = month(dayStart,  i_tz)
dom_c = dayofmonth(dayStart,  i_tz)
dow_c = dayofweek(dayStart,   i_tz)

// --- Cycle Calculations ---

// Yearly Cycle (Quarters Q1-Q4)
qYear = mo_c <= 3 ? 1 : mo_c <= 6 ? 2 : mo_c <= 9 ? 3 : 4

// Quarterly Cycle (Sub-Quarters)
qs_m    = math.floor((mo_c - 1) / 3) * 3 + 1
qs_t    = timestamp(i_tz, yy_c, qs_m, 1, start_hour, 0)
qe_t    = timestamp(i_tz, yy_c, qs_m + 3, 1, start_hour, 0)
q_dur   = qe_t - qs_t
q2Month = int(math.floor((time - qs_t) / q_dur * 4)) + 1
q2Month := math.max(1, math.min(4, q2Month))
// Monthly Cycle (Weeks) - Daye's Quarterly Theory (First Monday of the Month)
monday_time = dayStart - (dow_c == dayofweek.sunday ? -1 : dow_c - 2) * msDay
mon_y = year(monday_time, i_tz)
mon_m = month(monday_time, i_tz)
mon_d = dayofmonth(monday_time, i_tz)

// Timezone-safe mathematical calculation of the first Monday of the calendar month
dow_1st = ((1 - (mon_d - 1) % 7 + 7) % 7) + 1
dow_mon_1st = dow_1st // preserve compatibility if referenced elsewhere

first_monday_mon_m = dow_1st == dayofweek.monday ? 1 :
                      dow_1st == dayofweek.tuesday ? 7 :
                      dow_1st == dayofweek.wednesday ? 6 :
                      dow_1st == dayofweek.thursday ? 5 :
                      dow_1st == dayofweek.friday ? 4 :
                      dow_1st == dayofweek.saturday ? 3 : 2

f_bias(bool _newq,_c1,_c2,_o2,hmax1,hmax2,lmin1,lmin2,_h0,_l0)=> 
    var bias  = 0
    var bias1 = 0
    if _newq
        bias1 := bias[1]
        bias  := 0
        if _c1 > hmax2 
            bias := 1
        if _c1 < lmin2
            bias := -1
        if _c1 < hmax2 and _c1 > lmin2 and hmax1 > hmax2 and lmin1 > lmin2
            bias := -1
        if _c1 > lmin2 and _c1 < hmax2 and hmax1 < hmax2 and lmin1 < lmin2
            bias := 1
        if hmax1 <= hmax2 and lmin1 >= lmin2 
            bias := bias1 
        if hmax1 > hmax2 and _c1 < hmax2 and lmin1 < lmin2 and _c1 > lmin2 //and _o2 > _c2 
            bias := 0 
        if hmax1 > hmax2 and _c1 < hmax2 and lmin1 < lmin2 and _c1 > lmin2 //and _o2 < _c2 
            bias := 0   
    if bias == 1 and _l0 <= lmin1 
        bias := 2 
    if bias == -1 and _h0 >= hmax1 
        bias := -2           

    bias
