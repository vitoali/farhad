// ── Core Calculations ───────────────────────────────────────────────────────
string clean_ticker = syminfo.prefix + ":" + syminfo.ticker
float atr_val  = ta.atr(atr_len)
float vol_avg  = ta.sma(volume, imb_len)
float body     = math.abs(close - open)
float rng      = high - low
float body_rat = rng > 0.0 ? body / rng : 0.0
float vol_rat  = vol_avg > 0.0 ? volume / vol_avg : 1.0
bool  bull_body = close > open
bool  bear_body = close < open
float mom_fast  = ta.sma(close, 5)
float mom_slow  = ta.sma(close, imb_len * 2)
bool  bull_trend = mom_fast > mom_slow
bool  bear_trend = mom_fast < mom_slow
color neutral_col = color.rgb(175,160,120)

// ── Imbalance & Signal Detection ────────────────────────────────────────────
bool  vol_spike  = vol_rat >= vol_mult
bool  imb_bull   = bull_body and body_rat >= imb_thresh and vol_spike and bull_trend
bool  imb_bear   = bear_body and body_rat >= imb_thresh and vol_spike and bear_trend
bool  prev_bull  = imb_bull[1]
bool  prev_bear  = imb_bear[1]
bool  sig_bull_raw = imb_bull[1] and not imb_bull[2]
bool  sig_bear_raw = imb_bear[1] and not imb_bear[2]
float swing_lo_ref   = ta.lowest(low, imb_len)[4]
float swing_hi_ref   = ta.highest(high, imb_len)[4]
bool  liq_swept_bull = math.min(low[1], low[2], low[3]) < swing_lo_ref and close[1] > swing_lo_ref
bool  liq_swept_bear = math.max(high[1], high[2], high[3]) > swing_hi_ref and close[1] < swing_hi_ref
bool  liq_ok_bull    = not use_liq_sweep or liq_swept_bull
bool  liq_ok_bear    = not use_liq_sweep or liq_swept_bear
bool  in_london = not na(time(timeframe.period, "0700-1030:1234567", "UTC")[1])
bool  in_ny     = not na(time(timeframe.period, "1230-1500:1234567", "UTC")[1])
bool  sess_ok   = not use_session or ((sess_london and in_london) or (sess_ny and in_ny))
float htf_close_raw = request.security(syminfo.tickerid, htf_tf, close[1],             lookahead=barmerge.lookahead_on)
float htf_ma_raw    = request.security(syminfo.tickerid, htf_tf, ta.sma(close, 20)[1], lookahead=barmerge.lookahead_on)
bool  htf_bull  = htf_close_raw > htf_ma_raw
bool  htf_bear  = htf_close_raw < htf_ma_raw
string htf_trend_str = htf_bull ? "▲ Bull" : htf_bear ? "▼ Bear" : "◆ Neut"
color  htf_trend_col = htf_bull ? bull_col : htf_bear ? bear_col : neutral_col
float rsi_val = ta.rsi(close, 14)
bool  bull_div = low[1] < low[1 + imb_len] and rsi_val[1] > rsi_val[1 + imb_len]
bool  bear_div = high[1] > high[1 + imb_len] and rsi_val[1] < rsi_val[1 + imb_len]
bool  div_ok_bull  = not use_div      or bull_div
bool  div_ok_bear  = not use_div      or bear_div
bool  htf_ok_bull  = not use_mtf_align or htf_bull
bool  htf_ok_bear  = not use_mtf_align or htf_bear

bool  sig_bull = sig_bull_raw and liq_ok_bull and sess_ok and div_ok_bull and htf_ok_bull
bool  sig_bear = sig_bear_raw and liq_ok_bear and sess_ok and div_ok_bear and htf_ok_bear

// ── Persistent State ────────────────────────────────────────────────────────
var float entry_bull   = na

// ── Drawing Logic ───────────────────────────────────────────────────────────
int ext = bar_index + imb_len * 5
if use_ext_box
    if bull_active and not na(sl_bull) and low <= sl_bull
        bull_active := false
    if bear_active and not na(sl_bear) and high >= sl_bear
        bear_active := false
    if bull_active and not na(bx_zone_b)
        box.set_right(bx_zone_b, bar_index + imb_len * 5)
    if bear_active and not na(bx_zone_s)
        box.set_right(bx_zone_s, bar_index + imb_len * 5)
if sig_bull
    entry_bull  := close[1]
    sl_bull     := close[1] - sl_atr * atr_val[1]
    float rb     = entry_bull - sl_bull
    tp1_bull    := entry_bull + tp1_r * rb
    tp2_bull    := entry_bull + tp2_r * rb
    tp3_bull    := entry_bull + tp3_r * rb
    zone_hi_b   := entry_bull + atr_val[1] * 0.35
    zone_lo_b   := entry_bull - atr_val[1] * 0.35
    bull_active := true
    bear_active := false
    line.delete(ln_entry_b)
    line.delete(ln_sl_b)
    line.delete(ln_tp1_b)
    line.delete(ln_tp2_b)
    line.delete(ln_tp3_b)
    box.dele
